```javascript
class ContactImporterModal extends Component {
  state = {
    csvData: null,
    columnMapping: {},
    validationErrors: [],
    previewRows: []
  };

  async processCSV(file) {
    const data = await parseCSV(file);

    // Auto-detect columns
    const autoMapping = this.detectColumns(data.headers);

    // Validate data
    const errors = this.validateContacts(data.rows, autoMapping);

    this.setState({
      csvData: data,
      columnMapping: autoMapping,
      validationErrors: errors,
      previewRows: data.rows.slice(0, 5)
    });
  }

  detectColumns(headers) {
    const mapping = {};
    headers.forEach(header => {
      const lower = header.toLowerCase();
      if (lower.includes('email')) mapping.email = header;
      if (lower.includes('first') && lower.includes('name')) mapping.firstName = header;
      if (lower.includes('company')) mapping.company = header;
      // ... more auto-detection
    });
    return mapping;
  }
}
```

#### **4.1 Video Generation Engine**
```javascript
class VideoGenerationEngine {
  async generatePersonalizedVideos(baseVideo, contacts, options) {
    const results = [];
    const batchSize = options.concurrentJobs || 3;

    // Process in batches
    for (let i = 0; i < contacts.length; i += batchSize) {
      const batch = contacts.slice(i, i + batchSize);

      const batchPromises = batch.map(contact =>
        this.generateSingleVideo(baseVideo, contact, options)
      );

      const batchResults = await Promise.allSettled(batchPromises);
      results.push(...batchResults);

      // Update progress
      this.onProgress?.(results.length, contacts.length);
    }

    return results;
  }

  async generateSingleVideo(baseVideo, contact, options) {
    try {
      // 1. Replace tokens in video
      const personalizedVideo = await this.applyPersonalization(baseVideo, contact);

      // 2. Generate custom audio if needed
      const customAudio = await this.generateCustomAudio(contact, options);

      // 3. Apply audio to video
      const videoWithAudio = await this.applyAudio(personalizedVideo, customAudio);

      // 4. Generate thumbnail
      const thumbnail = await this.generateThumbnail(videoWithAudio);

      // 5. Create landing page
      const landingPage = await this.createLandingPage(contact, videoWithAudio, options);

      return {
        contact,
        video: videoWithAudio,
        thumbnail,
        landingPage,
        status: 'completed'
      };

    } catch (error) {
      return {
        contact,
        error: error.message,
        status: 'failed'
      };
    }
  }
}
```

**Features:**
- ✅ **Batch Processing**: 3-5 videos simultaneously

```javascript
      const personalizedScript = replaceTokens(baseVideo.script, contact);
```

```javascript
  constructor() {
    this.baseUrl = 'https://pageshot.site';
  }

  async captureWebsite(options) {
    const { url, width = 1920, height = 1080, format = 'png' } = options;
    
    // PageShot API endpoint (FREE, no API key needed!)
    const response = await fetch(
      `${this.baseUrl}/screenshot?url=${encodeURIComponent(url)}&width=${width}&height=${height}&format=${format}`
    );
    
    const blob = await response.blob();
    return URL.createObjectURL(blob);
  }

  async captureLinkedIn(linkedinUrl) {
    // Extract company name from LinkedIn URL
    const companyName = this.extractCompanyFromLinkedIn(linkedinUrl);
    
    // Try company website first
    const companyWebsite = await this.findCompanyWebsite(companyName);
    
    if (companyWebsite) {
      return await this.captureWebsite({ url: companyWebsite });
    }
    
    // Fallback to LinkedIn profile screenshot
    return await this.captureWebsite({ url: linkedinUrl });
  }

  async captureWithSmartCrop(url) {
    const fullScreenshot = await this.captureWebsite({ url, width: 1920, height: 1080 });
    
    // Smart crop to content area (skip headers, footers, sidebars)
    const cropped = await this.smartCrop(fullScreenshot, {
      targetWidth: 1920,
      targetHeight: 1080,
      cropMode: 'content' // 'full', 'content', 'hero'
    });
    
    return cropped;
  }
}
```

```javascript
// lib/engines/backgroundGenerator.js
export class BackgroundGenerator {
  async generateForContact(contact) {
    const backgrounds = [];

    // Method 1: Website screenshot
    if (contact.website) {
      const websiteScreenshot = await this.captureWebsite(contact.website);
      backgrounds.push({
        type: 'website',
        url: websiteScreenshot,
        priority: 1
      });
    }

    // Method 2: LinkedIn screenshot
    if (contact.linkedin) {
      const linkedinScreenshot = await this.captureLinkedIn(contact.linkedin);
      backgrounds.push({
        type: 'linkedin',
        url: linkedinScreenshot,
        priority: 2
      });
    }

    // Method 3: AI-generated industry background
    const industryBackground = await this.generateIndustryBackground(
      contact.industry || 'business'
    );
    backgrounds.push({
      type: 'ai-generated',
      url: industryBackground,
      priority: 3
    });

    // Method 4: Brand-colored gradient
    if (contact.website) {
      const brandColors = await this.extractBrandColors(contact.website);
      const gradientBackground = await this.createGradientBackground(brandColors);
      backgrounds.push({
        type: 'gradient',
        url: gradientBackground,
        priority: 4
      });
    }

    // Return highest priority available
    return backgrounds.sort((a, b) => a.priority - b.priority)[0].url;
  }

  async captureWebsite(url) {
    const service = new PageShotService();
    return await service.captureWithSmartCrop(url);
  }

  async generateIndustryBackground(industry) {
    // Use Stable Diffusion or template-based generation
    const templates = {
      'saas': 'gradient-blue-purple',
      'marketing': 'gradient-orange-pink',
      'finance': 'gradient-green-teal',
      'healthcare': 'gradient-blue-green',
      'education': 'gradient-purple-pink'
    };

    return await this.applyTemplate(templates[industry] || 'gradient-blue-purple');
  }

  async extractBrandColors(website) {
    // Use color-thief or similar to extract dominant colors
    const colors = await this.getColorPalette(website);
    return colors.slice(0, 3); // Top 3 brand colors
  }
}
```

```javascript
// lib/services/batchBackgroundProcessor.js
export class BatchBackgroundProcessor {
  async processContacts(contacts) {
    const results = [];
    const generator = new BackgroundGenerator();

    // Process in batches to avoid rate limits
    const batchSize = 5;
    for (let i = 0; i < contacts.length; i += batchSize) {
      const batch = contacts.slice(i, i + batchSize);
      
      const batchResults = await Promise.all(
        batch.map(async (contact) => {
          try {
            const background = await generator.generateForContact(contact);
            return { contact, background, success: true };
          } catch (error) {
            return { contact, background: null, success: false, error };
          }
        })
      );

      results.push(...batchResults);
      
      // Small delay between batches
      await this.delay(1000);
    }

    return results;
  }
}
```
