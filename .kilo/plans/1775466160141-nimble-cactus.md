```javascript
/**
 * Base Component Class - Vanilla JS replacement for React.Component
 */
import { escapeHtml } from '../../utils/sanitize.js';

export class Component {
  constructor(props = {}) {
    this.props = this.validateProps(props);
    this.state = {};
    this.element = null;
    this.children = [];
    this.eventListeners = [];
    this.subscriptions = [];
    this.documentListeners = [];
    this.isMounted = false;
    this.error = null;
    this.timers = [];
  }

  validateProps(props) {
    if (!props || typeof props !== 'object') return {};
    return { ...props };
  }

  render() {
    throw new Error('Subclasses must implement render() method');
  }

  mount(container) {
    if (!container || !(container instanceof HTMLElement)) {
      throw new Error('Container must be a valid HTMLElement');
    }

    try {
      this.element = this.render();
      if (!(this.element instanceof HTMLElement)) {
        throw new Error('render() must return an HTMLElement');
      }

      container.appendChild(this.element);
      this.isMounted = true;
      this.setupEventListeners();
      this.onMount();

      return this.element;
    } catch (error) {
      this.handleRenderError(error);
      throw error;
    }
  }

  handleRenderError(error) {
    console.error('Component render error:', error);
    this.error = error;
    
    if (this.element?.parentNode) {
      const errorDiv = document.createElement('div');
      errorDiv.className = 'component-error';
      errorDiv.setAttribute('role', 'alert');
      errorDiv.innerHTML = `
        <h3>Something went wrong</h3>
        <p>${escapeHtml(error.message)}</p>
      `;
      this.element.parentNode.replaceChild(errorDiv, this.element);
    }
  }

  unmount() {
    if (!this.isMounted) return;
    
    try {
      this.onUnmount();
      this.cleanupEventListeners();
      this.cleanupDocumentListeners();
      this.unsubscribeAll();
      this.cleanupTimers();

      if (this.element && this.element.parentNode) {
        this.element.parentNode.removeChild(this.element);
      }

      this.element = null;
      this.isMounted = false;
      this.error = null;
    } catch (error) {
      console.error('Error during unmount:', error);
    }
  }

  setState(newState, callback) {
    if (typeof newState === 'function') {
      this.state = newState(this.state);
    } else {
      this.state = { ...this.state, ...newState };
    }
    
    if (this.isMounted) {
      try {
        this.update();
      } catch (error) {
        this.handleRenderError(error);
      }
    }

    if (callback && typeof callback === 'function') {
      callback();
    }
  }

  update(newProps = {}) {
    const oldProps = this.props;
    this.props = this.validateProps({ ...this.props, ...newProps });

    if (this.shouldUpdate(oldProps, this.props)) {
      this.cleanupEventListeners();
      const newElement = this.render();
      if (this.element && this.element.parentNode) {
        this.element.parentNode.replaceChild(newElement, this.element);
      }
      this.element = newElement;
      this.setupEventListeners();
      this.onUpdate();
    }
  }

  shouldUpdate(oldProps, newProps) {
    return true;
  }

  setupEventListeners() {
    // Override in subclasses
  }

  cleanupEventListeners() {
    this.eventListeners.forEach(({ element, event, handler }) => {
      try {
        element.removeEventListener(event, handler);
      } catch (error) {
        console.warn('Failed to remove event listener:', error);
      }
    });
    this.eventListeners = [];
  }

  cleanupDocumentListeners() {
    this.documentListeners.forEach(({ element, event, handler }) => {
      try {
        element.removeEventListener(event, handler);
      } catch (error) {
        console.warn('Failed to remove document listener:', error);
      }
    });
    this.documentListeners = [];
  }

  cleanupTimers() {
    this.timers.forEach(id => {
      clearTimeout(id);
      clearInterval(id);
    });
    this.timers = [];
  }

  addEventListener(element, event, handler, options = {}) {
    element.addEventListener(event, handler, options);
    this.eventListeners.push({ element, event, handler, options });
  }

  addDocumentListener(event, handler, options = {}) {
    document.addEventListener(event, handler, options);
    this.documentListeners.push({ element: document, event, handler, options });
  }

  subscribeToStore(store, callback) {
    if (store && typeof store.subscribe === 'function') {
      const unsubscribe = store.subscribe(callback);
      this.subscriptions.push(unsubscribe);
      return unsubscribe;
    }
    return null;
  }

  unsubscribeAll() {
    this.subscriptions.forEach(unsubscribe => {
      if (typeof unsubscribe === 'function') {
        try {
          unsubscribe();
        } catch (error) {
          console.warn('Failed to unsubscribe:', error);
        }
      }
    });
    this.subscriptions = [];
  }

  createElementFromHTML(html) {
    const div = document.createElement('div');
    div.innerHTML = html.trim();
    return div.firstChild;
  }

  safeInterpolate(template, values) {
    return template.replace(/\${(\w+)}/g, (match, key) => {
      return escapeHtml(String(values[key] ?? ''));
    });
  }

  querySelector(selector) {
    return this.element ? this.element.querySelector(selector) : null;
  }

  emit(eventName, detail = null) {
    if (this.element) {
      const event = new CustomEvent(eventName, { detail, bubbles: true });
      this.element.dispatchEvent(event);
    }
  }

  setTimer(fn, delay) {
    const timerId = setTimeout(() => {
      fn();
      const index = this.timers.indexOf(timerId);
      if (index > -1) this.timers.splice(index, 1);
    }, delay);
    this.timers.push(timerId);
    return timerId;
  }

  setInterval(fn, delay) {
    const intervalId = setInterval(fn, delay);
    this.timers.push(intervalId);
    return intervalId;
  }

  onMount() {}
  onUnmount() {}
  onUpdate() {}
}

export default Component;
```

```javascript
import { Component } from './base/Component.js';
import { getStore } from '../stores/base/Store.js';

export class Canvas extends Component {
  constructor(props = {}) {
    super(props);

    this.projectStore = getStore('projectStore', ProjectStore);
    this.uiStore = getStore('uiStore', UIStore);
    this.timelineStore = getStore('timelineStore', TimelineStore);

    this.state = {
      style: {},
      fontSize: 24,
      canvasHeight: 'calc(100vh - 200px)',
      aspectRatio: 16/9,
      isOver: false,
    };

    this.canvasRef = null;
    this.wrapperRef = null;

    this.marginLeft = 20;
    this.marginTop = 60;
    this.DEFAULT_VIDEO_WIDTH = 1920;
    this.DEFAULT_FONT_SIZE = 24;

    this.handleResize = this.handleResize.bind(this);
    this.handleDrop = this.handleDrop.bind(this);
    this.handleDragOver = this.handleDragOver.bind(this);
    this.handleDragLeave = this.handleDragLeave.bind(this);
    this.subscribeToStores();
  }

  subscribeToStores() {
    this.subscribeToStore(this.projectStore, () => this.updateFromStores());
    this.subscribeToStore(this.uiStore, () => this.updateFromStores());
    this.subscribeToStore(this.timelineStore, () => this.updateFromStores());
  }

  updateFromStores() {
    const projectState = this.projectStore.getState();
    const timelineState = this.timelineStore.getState();

    const ratio = projectState.item?.ratio || { width: 16, height: 9 };
    const newAspectRatio = ratio.width / ratio.height;
    const timelineHeight = timelineState.timelineHeight || 0;
    const newCanvasHeight = `calc(100vh - 200px - ${timelineHeight}px)`;

    this.setState({
      aspectRatio: newAspectRatio,
      canvasHeight: newCanvasHeight,
    });

    this.updateCanvasPosition();
  }

  onMount() {
    window.addEventListener('resize', this.handleResize);
    if (this.wrapperRef) {
      this.projectStore.setPopcorn(this.wrapperRef);
    }
    this.updateCanvasPosition();
    this.updateFontSize();
    this.setupDragAndDrop();
  }

  onUnmount() {
    window.removeEventListener('resize', this.handleResize);
  }

  setupDragAndDrop() {
    if (this.wrapperRef) {
      this.addEventListener(this.wrapperRef, 'dragover', this.handleDragOver);
      this.addEventListener(this.wrapperRef, 'dragleave', this.handleDragLeave);
      this.addEventListener(this.wrapperRef, 'drop', this.handleDrop);
    }
  }

  handleDragOver(e) {
    e.preventDefault();
    this.setState({ isOver: true });
  }

  handleDragLeave(e) {
    e.preventDefault();
    const rect = this.wrapperRef.getBoundingClientRect();
    if (e.clientX < rect.left || e.clientX > rect.right || 
        e.clientY < rect.top || e.clientY > rect.bottom) {
      this.setState({ isOver: false });
    }
  }

  handleDrop(e) {
    e.preventDefault();
    this.setState({ isOver: false });
    
    const rect = this.wrapperRef.getBoundingClientRect();
    const position = {
      top: ((e.clientY - rect.top) / rect.height) * 100,
      left: ((e.clientX - rect.left) / rect.width) * 100,
      dropped: true,
    };
    
    this.uiStore.toggleRightBlock();
    console.log('Dropped at position:', position);
  }

  handleResize() {
    this.updateCanvasPosition();
    this.updateFontSize();
  }

  updateCanvasPosition() {
    if (!this.canvasRef || !this.wrapperRef) return;

    const maxWidth = this.canvasRef.offsetWidth - (this.marginLeft * 2);
    const maxHeight = this.canvasRef.offsetHeight - (this.marginTop * 2);
    const sideIndent = (maxWidth - (maxHeight * this.state.aspectRatio)) / 2;

    const newStyle = sideIndent > 0
      ? { margin: `${this.marginTop}px ${sideIndent + this.marginLeft}px` }
      : { margin: `${((maxHeight - (maxWidth / this.state.aspectRatio))) / 2 + this.marginTop}px ${this.marginLeft}px` };

    this.setState({ style: newStyle });
  }

  updateFontSize() {
    if (this.wrapperRef) {
      const newFontSize = this.DEFAULT_FONT_SIZE * (this.wrapperRef.offsetWidth / this.DEFAULT_VIDEO_WIDTH);
      this.setState({ fontSize: newFontSize });
      this.setTimer(() => {
        this.projectStore.runTextfill();
        this.projectStore.runMapResize();
      }, 0);
    }
  }

  render() {
    const projectState = this.projectStore.getState();
    const uiState = this.uiStore.getState();

    const html = `
      <div class="stager-wrapper" style="height: ${this.state.canvasHeight}">
        ${uiState.checkboxLeft ? '<div class="guidelines-activation"></div>' : ''}
        ${projectState.retarget?.showed ? `
          <div class="personalizer-activation">
            <button class="toggle-personalizer-btn">Toggle Personalizer</button>
          </div>
        ` : ''}
        <div class="embed-wrapper" style="margin: ${this.state.style.margin || '60px 20px'}">
          ${uiState.hasGuidLines ? '<div class="guidelines"></div>' : ''}
          ${projectState.isLoadingSequencer ? '<div class="hover-loading"></div>' : ''}
          ${projectState.isLoadingSequencer ? `
            <div class="loading-spinner">
              <div class="spinner"></div>
            </div>
          ` : ''}
          <div class="video-container" style="font-size: ${this.state.fontSize}px;">
            <div id="video" class="video ${this.state.isOver ? 'video-active' : ''}"></div>
          </div>
        </div>
      </div>
    `;

    const element = this.createElementFromHTML(html);
    this.canvasRef = element;
    this.wrapperRef = element.querySelector('.video-container');

    this.setupEventListeners(element);
    return element;
  }

  setupEventListeners(element) {
    const wrapper = element.querySelector('.embed-wrapper');
    if (wrapper) {
      this.wrapperRef = wrapper;
      this.setupDragAndDrop();
    }

    const toggleBtn = element.querySelector('.toggle-personalizer-btn');
    if (toggleBtn) {
      this.addEventListener(toggleBtn, 'click', () => {
        this.projectStore.toggleViewPersonalizer();
      });
    }
  }
}

class ProjectStore {
  constructor() {
    this.state = {
      item: { ratio: { width: 16, height: 9 } },
      isLoadingSequencer: false,
      retarget: { showed: false }
    };
  }
  getState() { return this.state; }
  setPopcorn() {}
  runTextfill() {}
  runMapResize() {}
  toggleViewPersonalizer() {}
}

class UIStore {
  constructor() {
    this.state = {
      hasGuidLines: false,
      checkboxLeft: false,
      checkboxRight: false,
      isExpand: false,
      radioButtonBottom: false,
      isTimelineOpen: false,
      isCanvasPresent: true
    };
  }
  getState() { return this.state; }
  toggleRightBlock() {}
}

class TimelineStore {
  constructor() {
    this.state = { timelineHeight: 0 };
  }
  getState() { return this.state; }
}

export default Canvas;
```
